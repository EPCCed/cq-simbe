cmake_minimum_required(VERSION 4.0)
project(CQSimBEEmbeddedConsumer LANGUAGES C CXX)
if(NOT CQ_SIMBE_SOURCE_DIR)
  message(FATAL_ERROR "Set CQ_SIMBE_SOURCE_DIR")
endif()

set(CMAKE_DISABLE_FIND_PACKAGE_unity TRUE)
add_subdirectory("${CQ_SIMBE_SOURCE_DIR}" cq-simbe)
if(TARGET sqft OR TARGET test_host_env)
  message(FATAL_ERROR "Embedding cq-simbe unexpectedly enabled examples/tests")
endif()
if(CQ_SIMBE_ENABLE_INSTALL)
  message(FATAL_ERROR "Embedding cq-simbe unexpectedly enabled installation")
endif()
add_executable(embedded_consumer ../consumer/main.c)
target_link_libraries(embedded_consumer PRIVATE CQ-SIMBE::cq-simbe)
enable_testing()
add_test(NAME embedded_consumer COMMAND embedded_consumer)
set_tests_properties(embedded_consumer PROPERTIES TIMEOUT 60)
