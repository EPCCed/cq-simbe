cmake_minimum_required(VERSION 4.0)
project(CQSimBEFortranConsumer LANGUAGES C Fortran)
option(CONSUMER_ENABLE_CXX "Enable CXX for static dependency packages" OFF)
if(CONSUMER_ENABLE_CXX)
  enable_language(CXX)
endif()
find_package(CQ-SIMBE 0.2 CONFIG REQUIRED COMPONENTS Fortran)
add_executable(fortran_consumer main.f90)
set_target_properties(fortran_consumer PROPERTIES Fortran_PREPROCESS ON)
target_link_libraries(fortran_consumer PRIVATE CQ-SIMBE::fortran)
enable_testing()
add_test(NAME installed_fortran_consumer COMMAND fortran_consumer)
set_tests_properties(installed_fortran_consumer PROPERTIES TIMEOUT 60)
get_target_property(core_definitions CQ-SIMBE::cq-simbe INTERFACE_COMPILE_DEFINITIONS)
if("CQ_WITH_MPI_COMMS=1" IN_LIST core_definitions)
  add_test(NAME installed_fortran_consumer_mpi
    COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2
      ${MPIEXEC_PREFLAGS} $<TARGET_FILE:fortran_consumer> ${MPIEXEC_POSTFLAGS})
  set_tests_properties(installed_fortran_consumer_mpi PROPERTIES TIMEOUT 60)
endif()
